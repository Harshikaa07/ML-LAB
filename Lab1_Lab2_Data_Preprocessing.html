standard_columns = [
    'latitude',
    'longitude',
    'suspect_age',
    'victim_age',
    'num_arrests',
    'property_loss_usd'
]

for column in standard_columns:
    mean = df_standardized[column].mean()
    std = df_standardized[column].std()
    
    df_standardized[column] = (
        df_standardized[column] - mean
    ) / std
